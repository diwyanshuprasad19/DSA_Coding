def max_heapify(arr, i, n):
    largest = i
    left = 2 * i + 1
    right = 2 * i + 2

    if left < n and arr[left] > arr[largest]:
        largest = left

    if right < n and arr[right] > arr[largest]:
        largest = right

    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        max_heapify(arr, largest, n)


def convert_min_heap_to_max_heap(arr):
    n = len(arr)

    # Start from last non-leaf node and heapify
    for i in range((n - 2) // 2, -1, -1):
        max_heapify(arr, i, n)


def print_array(arr):
    print(" ".join(map(str, arr)))

🧪 Sample Input
arr = [3, 5, 9, 6, 8, 20, 10, 12, 18, 9]

print("Min Heap array:")
print_array(arr)

convert_min_heap_to_max_heap(arr)

print("Max Heap array:")
print_array(arr)

⏱ Time Complexity
Build Max Heap

Heapifying one node: O(log n)

Number of internal nodes: n / 2

🔑 Total Time Complexity
O(n)


Building a heap bottom-up is linear time (important interview fact!)

💾 Space Complexity

No extra data structures used

Recursive stack depth = heap height

O(log n)

📊 Final Complexity Summary
Operation	Time	Space
Heapify	O(log n)	O(log n)
Build Max Heap	O(n)	O(1) / O(log n)
Overall	O(n)	O(log n)
