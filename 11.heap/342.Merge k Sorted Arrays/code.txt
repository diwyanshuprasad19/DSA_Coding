Code:

#include <bits/stdc++.h>
using namespace std;

// Custom data structure to hold value and its position
class Data {
public:
    int value, row, col;
    Data(int v, int r, int c) {
        value = v;
        row = r;
        col = c;
    }
};

// Custom comparator for min-heap
struct myComp {
    bool operator()(Data &d1, Data &d2) {
        return d1.value > d2.value; // min-heap
    }
};

class Solution {
public:
    // GFG-style function to merge K sorted arrays
    vector<int> mergeKArrays(vector<vector<int>> &arr, int K) {
        vector<int> ans;
        priority_queue<Data, vector<Data>, myComp> pq;

        // Push the first element of each array into the heap
        for (int i = 0; i < K; i++) {
            if (!arr[i].empty()) {
                pq.push(Data(arr[i][0], i, 0));
            }
        }

        // Extract min and push next element from same row
        while (!pq.empty()) {
            Data d = pq.top(); pq.pop();
            ans.push_back(d.value);
            int r = d.row, c = d.col;
            if (c + 1 < arr[r].size()) {
                pq.push(Data(arr[r][c + 1], r, c + 1));
            }
        }
        return ans;
    }
};

int main() {
    // Sample input: 3 sorted arrays
    vector<vector<int>> arr = {
        {1, 4, 5},
        {1, 3, 4},
        {2, 6}
    };

    int K = arr.size();
    Solution sol;

    vector<int> merged = sol.mergeKArrays(arr, K);

    // Output the merged sorted array
    cout << "Merged array: ";
    for (int num : merged)
        cout << num << " ";
    cout << endl;

    return 0;
}

Time & Space Complexity
✅ Time Complexity: O(N log K)
Where N = total number of elements across all arrays

Each insertion/extraction in the heap takes log K time

We do it N times → O(N log K)

✅ Space Complexity: O(N + K)
Final output array = O(N)

Min-heap stores at most K elements = O(K)

Total = O(N + K)
