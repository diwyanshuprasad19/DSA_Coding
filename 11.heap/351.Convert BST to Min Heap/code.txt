✅ Python Function-Based Code (BST → Min Heap)
# Definition of a BST node
class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None


# Inorder traversal to collect sorted values
def inorder_traversal(root, arr):
    if root is None:
        return
    inorder_traversal(root.left, arr)
    arr.append(root.data)
    inorder_traversal(root.right, arr)


# Preorder traversal to overwrite node values
def preorder_fill(root, arr, index):
    if root is None:
        return index

    root.data = arr[index]
    index += 1

    index = preorder_fill(root.left, arr, index)
    index = preorder_fill(root.right, arr, index)

    return index


# Convert BST to Min Heap
def convert_bst_to_min_heap(root):
    inorder_arr = []
    inorder_traversal(root, inorder_arr)
    preorder_fill(root, inorder_arr, 0)


# Preorder print (to verify result)
def preorder_print(root):
    if root is None:
        return
    print(root.data, end=" ")
    preorder_print(root.left)
    preorder_print(root.right)

🧪 Sample Input (BST)
# Construct BST
root = Node(4)
root.left = Node(2)
root.right = Node(6)
root.left.left = Node(1)
root.left.right = Node(3)
root.right.left = Node(5)
root.right.right = Node(7)

convert_bst_to_min_heap(root)
preorder_print(root)

BST Structure
          4
        /   \
       2     6
      / \   / \
     1   3 5   7

📤 Sample Output (Preorder Traversal of Min Heap)
1 2 3 4 5 6 7


✔ Tree structure unchanged
✔ Min Heap property satisfied

🧠 Intuition (Simple)

Inorder traversal of BST → sorted values

Preorder traversal fills root first

Smallest value goes to root → Min Heap formed automatically
