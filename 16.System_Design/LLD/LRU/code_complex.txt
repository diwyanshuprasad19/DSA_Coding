Question

Implement an LRU cache with:

Expiration (TTL)

Priority-based eviction

LRU behavior

O(1) access using HashMap + Doubly Linked List

Eviction Rules

Remove expired items first

If still full → remove lowest priority

If multiple with same priority → remove least recently used

Python Implementation
import time


class Node:
    def __init__(self, key, value, expire_time=None, priority=0):
        self.key = key
        self.value = value
        self.expire_time = expire_time
        self.priority = priority
        self.prev = None
        self.next = None


class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}

        # Dummy head and tail
        self.head = Node(None, None, None, None)
        self.tail = Node(None, None, None, None)
        self.head.next = self.tail
        self.tail.prev = self.head

    # ---------------- Helper Methods ----------------

    def _remove(self, node):
        prev_node = node.prev
        next_node = node.next
        prev_node.next = next_node
        next_node.prev = prev_node

    def _add_to_front(self, node):
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node

    def _is_expired(self, node):
        return node.expire_time is not None and node.expire_time <= time.time()

    def _evict(self):
        # 1️⃣ Remove expired items first
        current = self.tail.prev
        while current != self.head:
            prev_node = current.prev
            if self._is_expired(current):
                self._remove(current)
                del self.cache[current.key]
            current = prev_node

        if len(self.cache) < self.capacity:
            return

        # 2️⃣ Evict lowest priority, LRU if tie
        node = self.tail.prev
        lowest_priority = float("inf")
        candidates = []

        while node != self.head:
            if not self._is_expired(node):
                if node.priority < lowest_priority:
                    lowest_priority = node.priority
                    candidates = [node]
                elif node.priority == lowest_priority:
                    candidates.append(node)
            node = node.prev

        # Remove LRU among lowest priority
        if candidates:
            lru_node = candidates[-1]  # least recently used
            self._remove(lru_node)
            del self.cache[lru_node.key]

    # ---------------- Public APIs ----------------

    def get(self, key):
        if key not in self.cache:
            return -1

        node = self.cache[key]

        if self._is_expired(node):
            self._remove(node)
            del self.cache[key]
            return -1

        # Move to front (most recently used)
        self._remove(node)
        self._add_to_front(node)
        return node.value

    def set(self, key, value, ttl=None, priority=0):
        expire_time = time.time() + ttl if ttl is not None else None

        if key in self.cache:
            node = self.cache[key]
            node.value = value
            node.expire_time = expire_time
            node.priority = priority
            self._remove(node)
            self._add_to_front(node)
        else:
            if len(self.cache) >= self.capacity:
                self._evict()

            node = Node(key, value, expire_time, priority)
            self.cache[key] = node
            self._add_to_front(node)
