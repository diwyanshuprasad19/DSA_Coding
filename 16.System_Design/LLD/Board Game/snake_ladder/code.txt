@dataclass(frozen=True)
class Snake:
    start: int
    end: int

    def __post_init__(self):

        if self.start <= self.end:
            raise InvalidSnakeException(
                "Snake start must be greater than end."
            )


# =========================================================
# 5. LADDER
# =========================================================

@dataclass(frozen=True)
class Ladder:
    start: int
    end: int

    def __post_init__(self):

        if self.start >= self.end:
            raise InvalidLadderException(
                "Ladder start must be smaller than end."
            )

# 6. DICE STRATEGY
# Strategy Pattern
# =========================================================

class DiceStrategy(ABC):

    @abstractmethod
    def roll(self):
        pass


class StandardDice(DiceStrategy):

    def __init__(self, sides=6, number_of_dice=1):

        if sides <= 0:
            raise SnakeLadderException(
                "Dice sides must be positive."
            )

        if number_of_dice <= 0:
            raise SnakeLadderException(
                "Number of dice must be positive."
            )

        self.sides = sides
        self.number_of_dice = number_of_dice


    def roll(self):

        total = 0

        for _ in range(self.number_of_dice):
            total += random.randint(
                1,
                self.sides
            )

        return total


# =========================================================
# 7. FIXED DICE
# Useful for testing
# =========================================================

class FixedDice(DiceStrategy):

    def __init__(self, values):
        self.values = values
        self.index = 0


    def roll(self):

        if self.index >= len(self.values):
            raise SnakeLadderException(
                "No more fixed dice values."
            )

        value = self.values[self.index]

        self.index += 1

        return value


# =========================================================
# 8. BOARD
# =========================================================

class Board:

    def __init__(self, size, jumps):

        if size <= 1:
            raise InvalidBoardException(
                "Board size must be greater than 1."
            )

        self.size = size

        # start_position -> Snake/Ladder
        self.jumps = jumps


    def resolve_position(self, position):

        visited = set()

        while position in self.jumps:

            # Detect loops like:
            # 10 -> 20
            # 20 -> 10

            if position in visited:
                raise BoardCycleException(
                    "Cycle detected in board."
                )

            visited.add(position)

            jump = self.jumps[position]

            if isinstance(jump, Snake):

                print(
                    f"Snake: "
                    f"{jump.start} -> {jump.end}"
                )

            elif isinstance(jump, Ladder):

                print(
                    f"Ladder: "
                    f"{jump.start} -> {jump.end}"
                )

            position = jump.end

        return position


# =========================================================
# 9. BOARD BUILDER
# Builder Pattern
# =========================================================

class BoardBuilder:

    def __init__(self, size=100):

        if size <= 1:
            raise InvalidBoardException(
                "Board size must be greater than 1."
            )

        self.size = size

        self.jumps = {}


    # -----------------------------------------------------
    # Add Snake
    # -----------------------------------------------------

    def add_snake(self, start, end):

        self._validate_position(start)
        self._validate_position(end)

        if start in self.jumps:
            raise InvalidBoardException(
                f"Cell {start} already contains a jump."
            )

        self.jumps[start] = Snake(
            start,
            end
        )

        return self


    # -----------------------------------------------------
    # Add Ladder
    # -----------------------------------------------------

    def add_ladder(self, start, end):

        self._validate_position(start)
        self._validate_position(end)

        if start in self.jumps:
            raise InvalidBoardException(
                f"Cell {start} already contains a jump."
            )

        self.jumps[start] = Ladder(
            start,
            end
        )

        return self


    # -----------------------------------------------------
    # Validate position
    # -----------------------------------------------------

    def _validate_position(self, position):

        if position < 1 or position > self.size:

            raise InvalidBoardException(
                f"Position must be between "
                f"1 and {self.size}."
            )


    # -----------------------------------------------------
    # Build Board
    # -----------------------------------------------------

    def build(self):

        board = Board(
            self.size,
            dict(self.jumps)
        )

        # Check for cycles
        for start in self.jumps:
            board.resolve_position(start)

        return board

class SnakeAndLadderGame:

    def __init__(
        self,
        board,
        players,
        dice
    ):

        if len(players) < 2:
            raise InvalidPlayerException(
                "At least 2 players are required."
            )

        self.board = board
        self.players = players
        self.dice = dice

        self.current_player_index = 0

        self.status = GameStatus.NOT_STARTED

        self.winner = None


    # -----------------------------------------------------
    # Start Game
    # -----------------------------------------------------

    def start(self):

        if self.status != GameStatus.NOT_STARTED:

            raise GameStateException(
                "Game has already started."
            )

        self.status = GameStatus.IN_PROGRESS

        print("Game started!")


    # -----------------------------------------------------
    # Current Player
    # -----------------------------------------------------

    def get_current_player(self):

        return self.players[
            self.current_player_index
        ]


    # -----------------------------------------------------
    # Play One Turn
    # -----------------------------------------------------

    def play_turn(self):

        if self.status == GameStatus.NOT_STARTED:

            raise GameStateException(
                "Game has not started."
            )


        if self.status == GameStatus.FINISHED:

            raise GameStateException(
                "Game has already finished."
            )


        # 1. Current player

        player = self.get_current_player()


        # 2. Roll dice

        dice_value = self.dice.roll()


        old_position = player.position


        print(
            f"\n{player.name} rolled "
            f"{dice_value}"
        )


        # 3. Calculate new position

        next_position = (
            player.position + dice_value
        )


        # 4. Overshoot

        if next_position > self.board.size:

            print(
                "Cannot move beyond board."
            )

        else:

            # 5. Resolve snake / ladder

            final_position = (
                self.board.resolve_position(
                    next_position
                )
            )

            player.position = final_position


        print(
            f"{old_position} -> "
            f"{player.position}"
        )


        # 6. Winner

        won = (
            player.position == self.board.size
        )


        if won:

            self.winner = player

            self.status = GameStatus.FINISHED

            print(
                f"{player.name} won!"
            )


        else:

            # 7. Next player
            self._move_to_next_player()


        return TurnResult(
            player_name=player.name,
            dice_value=dice_value,
            old_position=old_position,
            new_position=player.position,
            won=won
        )


    # -----------------------------------------------------
    # Next Player
    # -----------------------------------------------------

    def _move_to_next_player(self):

        self.current_player_index = (
            self.current_player_index + 1
        ) % len(self.players)

__post_init__() is a special method used with Python @dataclass.

Example:

from dataclasses import dataclass

@dataclass
class Snake:
    start: int
    end: int

    def __post_init__(self):
        if self.start <= self.end:
            raise InvalidSnakeException(
                "Snake start must be greater than end."
            )

Because of @dataclass, Python automatically creates the __init__() for you. Conceptually, it behaves like:

def __init__(self, start, end):
    self.start = start
    self.end = end

    self.__post_init__()

So when you write:

snake = Snake(90, 20)

the flow is:

Snake(90, 20)
    ↓
auto-generated __init__()
    ↓
self.start = 90
self.end = 20
    ↓
__post_init__()
    ↓
validation

Why not just use __init__()? You can. If you are not using @dataclass, write:

class Snake:

    def __init__(self, start, end):

        if start <= end:
            raise InvalidSnakeException(
                "Snake start must be greater than end."
            )

        self.start = start
        self.end = end

But with:

@dataclass

Python already generates __init__(), so __post_init__() is the convenient place for extra validation after fields are assigned.

pass in the exception class does not mean the message is ignored.

When you write:

class GameStateException(SnakeLadderException):
    pass

you are saying:

Create a new exception type called GameStateException, but don't add any extra behavior. Just inherit everything from the parent.

And the parent:

class SnakeLadderException(Exception):
    pass

inherits from Python's built-in Exception.

So this:

raise GameStateException(
    "Game has already started."
)

works because GameStateException ultimately inherits the built-in Exception.__init__().

Conceptually the inheritance is:

Exception
   ↑
SnakeLadderException
   ↑
GameStateException

So when you do:

GameStateException("Game has already started.")

Python basically uses the inherited constructor from Exception and stores that message inside the exception object.

Then raise means:

Stop normal execution here and throw this error.
