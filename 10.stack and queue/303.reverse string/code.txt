#include <iostream>
#include <stack>

using namespace std;

char* reverse(char *S, int len) {
    stack<char> s;

    // Push all characters onto the stack
    for (int i = 0; i < len; i++)
        s.push(S[i]);

    // Allocate memory for the reversed string
    char *a = new char[len + 1]; // +1 for null terminator
    int i = 0;

    // Pop characters from the stack to reverse the string
    while (!s.empty()) {
        a[i++] = s.top();
        s.pop();
    }
    
    a[i] = '\0'; // Null terminate the reversed string
    return a;
}

// Driver Code
int main() {
    char str[] = "GeeksForGeeks";
    int len = sizeof(str) / sizeof(str[0]) - 1; // Exclude null terminator

    char *reversedStr = reverse(str, len);
    cout << "Reversed String: " << reversedStr << endl;

    delete[] reversedStr; // Free dynamically allocated memory
    return 0;
}
