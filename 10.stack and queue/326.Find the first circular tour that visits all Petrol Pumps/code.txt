#include <bits/stdc++.h>
using namespace std;

// LeetCode Equivalent Solution
class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        int n = gas.size();
        int total_gas = 0, total_cost = 0;
        int curr_gas = 0, starting_point = 0;
        //first run a loop such that total gas at 
        //stating is greater than or equal to total spent gas
        for (int i = 0; i < n; i++) {
            // Track total fuel available and required
            total_gas += gas[i];
            total_cost += cost[i];

            // Current fuel balance
            curr_gas += gas[i] - cost[i];

            // If fuel runs out, reset starting point
            if (curr_gas < 0) {
                starting_point = i + 1;
                curr_gas = 0; // Reset fuel
            }
        }

        return (total_gas < total_cost) ? -1 : starting_point;     
    }
};

// GFG Equivalent Solution (Using Petrol Pump Struct)
struct petrolPump {
    int petrol;
    int distance;
};

int tour(petrolPump p[], int n) {
    int start = 0, fuel_tank = 0, fuel_shortage = 0;

    for (int i = 0; i < n; i++) {
        fuel_tank += p[i].petrol - p[i].distance;

        if (fuel_tank < 0) {
            start = i + 1;
            fuel_shortage += fuel_tank;
            fuel_tank = 0; // Reset fuel
        }
    }

    return (fuel_tank + fuel_shortage >= 0) ? start : -1;
}

// Driver Code
int main() {
    vector<int> gas = {4, 6, 7, 4};  
    vector<int> cost = {6, 5, 3, 5}; 

    Solution obj;
    cout << "LeetCode Solution Starting Index: " << obj.canCompleteCircuit(gas, cost) << endl;

    petrolPump p[] = {{4, 6}, {6, 5}, {7, 3}, {4, 5}};
    int n = sizeof(p) / sizeof(p[0]);

    cout << "GFG Solution Starting Index: " << tour(p, n) << endl;
    return 0;
}
