#include <iostream>
#include <stack>
using namespace std;

class QueueUsingTwoStacks {
    stack<int> s1, s2;

public:
    void push(int x) {
        s1.push(x);
    }

    int pop() {
        if (s2.empty()) {
            if (s1.empty()) {
                cout << "Queue Underflow\n";
                return -1;
            }
            while (!s1.empty()) {
                s2.push(s1.top());
                s1.pop();
            }
        }
        int frontElement = s2.top();
        s2.pop();
        return frontElement;
    }

    int front() {
        if (s2.empty()) {
            if (s1.empty()) {
                cout << "Queue is empty\n";
                return -1;
            }
            while (!s1.empty()) {
                s2.push(s1.top());
                s1.pop();
            }
        }
        return s2.top();
    }

    int size() {
        return s1.size() + s2.size();
    }
};

int main() {
    QueueUsingTwoStacks q;
    q.push(1);
    q.push(2);
    q.push(3);

    cout << "Front: " << q.front() << endl;  // Expected: 1
    cout << "Popped: " << q.pop() << endl;   // Expected: 1
    cout << "Size: " << q.size() << endl;    // Expected: 2

    return 0;
}

#include <iostream>
#include <stack>
using namespace std;

class QueueUsingOneStack {
    stack<int> s;

public:
    void push(int x) {
        s.push(x);
    }

    int pop() {
        if (s.empty()) {
            cout << "Queue Underflow\n";
            return -1;
        }

        int topElement = s.top();
        s.pop();

        if (s.empty()) return topElement; // Base case (front element)

        int frontElement = pop(); // Recursively remove until we reach front
        s.push(topElement); // Push back other elements
        return frontElement;
    }

    int front() {
        if (s.empty()) {
            cout << "Queue is empty\n";
            return -1;
        }

        int topElement = s.top();
        s.pop();

        if (s.empty()) {
            s.push(topElement); // Restore stack
            return topElement; // Base case (front element)
        }

        int frontElement = front();
        s.push(topElement);
        return frontElement;
    }

    int size() {
        return s.size();
    }
};

int main() {
    QueueUsingOneStack q;
    q.push(1);
    q.push(2);
    q.push(3);

    cout << "Front: " << q.front() << endl;  // Expected: 1
    cout << "Popped: " << q.pop() << endl;   // Expected: 1
    cout << "Size: " << q.size() << endl;    // Expected: 2

    return 0;
}
