#include <iostream>
#include <queue>

using namespace std;

// Recursive function to reverse a queue
void solve(queue<int> &q) {
    if (q.empty()) {
        return;
    }
    
    int x = q.front();  // Store front element
    q.pop();            // Remove front element
    
    solve(q);           // Recursive call to process the rest of the queue
    
    q.push(x);          // Push the stored element back at the end
}

// Function to reverse the queue
queue<int> rev(queue<int> q) {
    solve(q);
    return q;
}

// Driver function to test the code
int main() {
    queue<int> q;
    q.push(1);
    q.push(2);
    q.push(3);
    q.push(4);
    q.push(5);

    cout << "Original Queue: ";
    queue<int> temp = q;
    while (!temp.empty()) {
        cout << temp.front() << " ";
        temp.pop();
    }
    cout << endl;

    // Reverse the queue
    q = rev(q);

    cout << "Reversed Queue: ";
    while (!q.empty()) {
        cout << q.front() << " ";
        q.pop();
    }
    cout << endl;

    return 0;
}



Time and space


Time & Space Complexity Analysis:
Time Complexity:

Each element is removed (dequeued) once → O(N)
Each element is added (enqueued) back once → O(N)
So, the overall time complexity is O(N)
Space Complexity:

The recursion stack stores N elements in the worst case (since we recurse N times).
Recursive call stack takes O(N) space.
Queue uses O(N) space.
Total space complexity: O(N)
