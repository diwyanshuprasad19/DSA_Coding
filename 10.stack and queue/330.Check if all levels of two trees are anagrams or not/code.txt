#include <bits/stdc++.h>
using namespace std;

// Definition for a Binary Tree Node
struct Node {
    int data;
    Node* left, *right;
    Node(int val) {
        data = val;
        left = right = NULL;
    }
};

class Solution {
public:
    bool areAnagrams(Node* root1, Node* root2) {
        if (root1 == NULL && root2 == NULL) return true;
        if (root1 == NULL || root2 == NULL) return false;

        queue<Node*> q1, q2;
        q1.push(root1);
        q2.push(root2);

        while (!q1.empty()) {
            int size1 = q1.size();
            int size2 = q2.size();
            if (size1 != size2) return false;

            vector<int> v1, v2;

            while (size1--) {
                Node* node1 = q1.front(); q1.pop();
                Node* node2 = q2.front(); q2.pop();

                if (node1->left) q1.push(node1->left);
                if (node1->right) q1.push(node1->right);
                if (node2->left) q2.push(node2->left);
                if (node2->right) q2.push(node2->right);

                v1.push_back(node1->data);
                v2.push_back(node2->data);
            }

            // Sort both level values to check for anagram
            sort(v1.begin(), v1.end());
            sort(v2.begin(), v2.end());

            if (v1 != v2) return false;
        }
        return true;
    }
};

// Driver Code
int main() {
    // Tree 1
    Node* root1 = new Node(1);
    root1->left = new Node(3);
    root1->right = new Node(2);
    root1->left->left = new Node(5);
    root1->left->right = new Node(4);
    
    // Tree 2
    Node* root2 = new Node(1);
    root2->left = new Node(2);
    root2->right = new Node(3);
    root2->left->left = new Node(4);
    root2->left->right = new Node(5);

    Solution obj;
    bool result = obj.areAnagrams(root1, root2);

    cout << (result ? "Yes" : "No") << endl;

    return 0;
}


TIme and space:

Time & Space Complexity Analysis
Time Complexity:
BFS Traversal: O(N)
Sorting each level: O(k log k), where k is the number of nodes in that level.
In the worst case (balanced binary tree), k = O(N), so sorting takes O(N log N).
Total Time Complexity: O(N log k) (where k is max nodes in a level).
Space Complexity:
Queue storage: O(N) (for level-wise BFS traversal).
Vector storage: O(N) (for storing values at each level).
Total Space Complexity: O(N).
