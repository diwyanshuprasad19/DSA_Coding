The given solution uses BFS (Breadth-First Search) to solve the Rotting Oranges problem.


Code:

#include <iostream>
#include <vector>
#include <queue>

using namespace std;

class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        if (grid.empty()) return 0;

        int rows = grid.size(), cols = grid[0].size();
        int days = 0, count = 0, totalOranges = 0;
        queue<pair<int, int>> q;

        // Step 1: Count fresh oranges and push rotten oranges into queue
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (grid[i][j] != 0) totalOranges++; // Count total non-empty cells
                if (grid[i][j] == 2) q.push({i, j}); // Store initial rotten oranges
            }
        }

        // Direction vectors for moving up, left, down, and right
        int dx[4] = {0, -1, 0, 1};
        int dy[4] = {1, 0, -1, 0};

        // Step 2: Perform BFS to rot fresh oranges
        while (!q.empty()) {
            int size = q.size();
            count += size; // Count number of rotten oranges in queue

            while (size--) {
                int x = q.front().first;
                int y = q.front().second;
                q.pop();

                // Traverse in all 4 directions
                for (int i = 0; i < 4; i++) {
                    int newX = x + dx[i];
                    int newY = y + dy[i];

                    // Check if within bounds and fresh orange is present
                    if (newX < 0 || newX >= rows || newY < 0 || newY >= cols || grid[newX][newY] != 1)
                        continue;

                    // Mark orange as rotten and add to queue
                    grid[newX][newY] = 2;
                    q.push({newX, newY});
                }
            }

            // If queue still has elements, increment days
            if (!q.empty()) days++;
        }

        // Step 3: Return days if all oranges are rotten, else return -1
        return (totalOranges == count) ? days : -1;
    }
};

// Driver Code
int main() {
    Solution sol;
    vector<vector<int>> grid = {
        {2, 1, 1},
        {1, 1, 0},
        {0, 1, 1}
    };

    cout << "Minimum number of days: " << sol.orangesRotting(grid) << endl;
    return 0;
}



Time and space comeplxity:

Complexity Analysis
Time Complexity: O(n * m)
We traverse the entire grid once to count oranges O(n * m).
BFS processes each cell at most once O(n * m).
Overall complexity: O(n * m).
Space Complexity: O(n * m)
The queue (q) can store at most O(n * m) elements in the worst case (when all oranges are rotten at different times).
We modify the grid in-place, so no extra space is used apart from the queue.
Overall space complexity: O(n * m).
