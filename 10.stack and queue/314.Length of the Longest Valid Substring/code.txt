#include <iostream>
#include <string>
#include <algorithm>

using namespace std;

int findMaxLen(string s) {
    int left = 0, right = 0, ans = 0, n = s.size();

    // Left to Right Pass
    for (int i = 0; i < n; i++) {
        if (s[i] == '(') left++;
        else right++;

        if (left == right)
            ans = max(ans, left + right);
        else if (right > left)
            left = right = 0; // Reset if invalid
    }

    left = 0, right = 0;

    // Right to Left Pass
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == '(') left++;
        else right++;

        if (left == right)
            ans = max(ans, left + right);
        else if (left > right)
            left = right = 0; // Reset if invalid
    }

    return ans;
}

// Driver Code
int main() {
    string s = "(()))())(";
    cout << "Longest Valid Parentheses Length: " << findMaxLen(s) << endl;
    return 0;
}
