class NextGreaterElement {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        int n = nums.size();
        vector<int> nge(n, -1);  // Initialize result with -1
        stack<int> st;  // Stack to store indexes

        for (int i = 0; i < n; i++) {
            // Maintain a decreasing order in stack
            while (!st.empty() && nums[st.top()] < nums[i]) {
                nge[st.top()] = nums[i];
                st.pop();
            }
            st.push(i);
        }
        return nge;
    }
};

int main() {
    NextGreaterElement ngeSolver;
    vector<int> nums = {4, 5, 2, 10, 8};

    vector<int> result = ngeSolver.nextGreaterElements(nums);
    
    cout << "Next Greater Elements: ";
    for (int val : result) {
        cout << val << " ";
    }
    cout << endl;

    return 0;
}
