#include <bits/stdc++.h>
using namespace std;

class Stack {
    int top;
    int *arr;
    int size;

public:
    Stack() {
        top = -1;
        size = 1000; // Default size
        arr = new int[size];
    }

    void push(int x) {
        if (top >= size - 1) {
            cout << "Stack Overflow" << endl;
            return;
        }
        top++;
        arr[top] = x;
    }

    int pop() {
        if (top == -1) {
            cout << "Stack Underflow" << endl;
            return -1;
        }
        int x = arr[top];
        top--;
        return x;
    }

    int Top() {
        if (top == -1) {
            cout << "Stack is empty" << endl;
            return -1;
        }
        return arr[top];
    }

    int Size() {
        return top + 1;
    }
};

int main() {
    Stack s;
    s.push(6);
    s.push(7);
    s.push(8);

    cout << "Popped: " << s.pop() << endl;   // Expected output: 8
    cout << "Top: " << s.Top() << endl;     // Expected output: 7
    cout << "Size: " << s.Size() << endl;   // Expected output: 2

    return 0;
}
