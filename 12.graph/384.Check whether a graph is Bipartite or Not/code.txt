✅ Python Code — Check if Graph is Bipartite (BFS)
from collections import deque

def check(start, adj, color):
    q = deque()
    q.append(start)
    color[start] = 0

    while q:
        node = q.popleft()

        for it in adj[node]:
            # If the node is not colored
            if color[it] == -1:
                color[it] = 1 - color[node]
                q.append(it)
            # If adjacent node has same color
            elif color[it] == color[node]:
                return False

    return True


def isBipartite(V, adj):
    color = [-1] * V

    for i in range(V):
        if color[i] == -1:
            if not check(i, adj, color):
                return False

    return True

▶️ Sample Input
V = 4
adj = [[] for _ in range(V)]

edges = [
    (0, 2),
    (0, 3),
    (2, 3),
    (3, 1)
]

for u, v in edges:
    adj[u].append(v)
    adj[v].append(u)

print(isBipartite(V, adj))

❌ Sample Output
False

Why?

Graph contains an odd-length cycle:

0 → 2 → 3 → 0


So it is not bipartite.
