Python Equivalent – Floyd–Warshall Algorithm

def shortest_distance(matrix):
    n = len(matrix)
    INF = int(1e9)

    # Step 1: Preprocessing
    for i in range(n):
        for j in range(n):
            if matrix[i][j] == -1:
                matrix[i][j] = INF
            if i == j:
                matrix[i][j] = 0

    # Step 2: Floyd–Warshall core logic
    for k in range(n):
        for i in range(n):
            for j in range(n):
                matrix[i][j] = min(
                    matrix[i][j],
                    matrix[i][k] + matrix[k][j]
                )

    # Step 3: Convert INF back to -1
    for i in range(n):
        for j in range(n):
            if matrix[i][j] == INF:
                matrix[i][j] = -1

🧪 Sample Input (Same as C++)
V = 4
matrix = [[-1]*V for _ in range(V)]

# first vertex, adjacent vertex, weight
matrix[0][1] = 2
matrix[1][0] = 1
matrix[1][2] = 3
matrix[3][0] = 3
matrix[3][1] = 5
matrix[3][2] = 4

shortest_distance(matrix)

for row in matrix:
    print(*row)
