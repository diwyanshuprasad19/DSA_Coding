Code:

#include <bits/stdc++.h>
using namespace std;

struct Node {
    int data;
    Node* left, *right;
    
    Node(int val) {
        data = val;
        left = right = NULL;
    }
};

vector<int> solve(Node* root) {
    if (root == NULL) return {1, 0, INT_MAX, INT_MIN}; // {isBST, size, minVal, maxVal}
    if (root->left == NULL && root->right == NULL) return {1, 1, root->data, root->data};

    vector<int> l = solve(root->left);
    vector<int> r = solve(root->right);

    if (l[0] && r[0]) { // If both left and right subtrees are BST
        if (l[3] < root->data && r[2] > root->data) { // Checking BST condition
            int x = (l[2] == INT_MAX) ? root->data : l[2];
            int y = (r[3] == INT_MIN) ? root->data : r[3];
            return {1, l[1] + r[1] + 1, x, y}; // Updated BST properties
        }
    }
    return {0, max(l[1], r[1]), l[2], r[3]}; // If not a BST, return max BST size found
}

int largestBst(Node* root) {
    vector<int> ans = solve(root);
    return ans[1]; // Returning the size of the largest BST
}

// Driver function (example usage)
int main() {
    Node* root = new Node(10);
    root->left = new Node(5);
    root->right = new Node(15);
    root->left->left = new Node(1);
    root->left->right = new Node(8);
    root->right->right = new Node(7); // Invalid BST node

    cout << "Size of the largest BST: " << largestBst(root) << endl;
    return 0;
}


Time and Space Complexity
Time Complexity:
Each node is visited once and performs O(1) work.
Total complexity = O(N), where N is the number of nodes.
Space Complexity:
Uses O(H) space due to recursive calls (where H is tree height).
Best case (Balanced BST): O(log N)
Worst case (Skewed Tree): O(N)
