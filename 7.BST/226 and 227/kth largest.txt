#include <bits/stdc++.h>
using namespace std;

// Structure of a BST Node
struct Node {
    int data;
    Node* left, *right;
    
    Node(int val) {
        data = val;
        left = right = NULL;
    }
};

int ans;  // Variable to store the Kth largest element

// Function to find Kth largest element using reverse inorder traversal
void solve(Node* root, int k, int &idx) {
    if (root == NULL) return;

    // Traverse right subtree first (reverse inorder)
    solve(root->right, k, idx);

    // Check if the current node is the Kth largest
    if (k == idx) {
        ans = root->data;
        idx++;
        return;
    }
    else idx++;

    // Traverse left subtree
    solve(root->left, k, idx);
}

// Wrapper function to find Kth largest element
int kthLargest(Node *root, int K) {
    int idx = 1;
    ans = -1;  // Default value in case K is out of bounds
    solve(root, K, idx);
    return ans;
}

// Utility function to insert nodes into the BST
Node* insert(Node* root, int key) {
    if (root == NULL) return new Node(key);
    
    if (key < root->data)
        root->left = insert(root->left, key);
    else
        root->right = insert(root->right, key);
    
    return root;
}

// Driver code
int main() {
    Node* root = NULL;
    root = insert(root, 20);
    root = insert(root, 10);
    root = insert(root, 30);
    root = insert(root, 5);
    root = insert(root, 15);
    root = insert(root, 25);
    root = insert(root, 35);

    int K = 3; // Find 3rd largest element
    cout << "The " << K << "rd largest element is: " << kthLargest(root, K) << endl;

    return 0;
}
