class Solution:
    def hIndex(self, citations: List[int]) -> int:
        n =len(citations)
        citations.sort()
        ans=0
        for i,v in enumerate(citations):
            if n-i<=v:
                return n-i
        
        return 0

General format:

range(start, stop, step)

So here:

range(n, -1, -1)

means:

start = n
stop = -1   # stop before -1
step = -1   # move backwards

Example:

n = 5

for i in range(n, -1, -1):
    print(i)

Output:

5
4
3
2
1
0

Important: Python range does not include stop value.

Sol2:

class Solution:
    def hIndex(self, citations: List[int]) -> int:
        n = len(citations)
        temp = [0 for _ in range(n + 1)]

        for i,v in enumerate(citations):
            if v > n :
                temp[n] += 1
            else:
                temp[v] += 1
        
        total = 0
        for i in range(n, -1, -1):
            total += temp[i]
            if total >= i:
                return i

TC:O(n)
SC:O(1)
