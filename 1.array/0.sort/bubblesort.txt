#include <iostream>
#include <vector>
using namespace std;

// Function to perform Bubble Sort
void bubbleSort(vector<int>& arr) {
    int n = arr.size();
    // Outer loop for each pass
    for (int i = 0; i < n - 1; i++) {
        // Inner loop for comparing adjacent elements
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                // Swap if elements are in the wrong order
                swap(arr[j], arr[j + 1]);
            }
        }
    }
}

int main() {
    vector<int> arr = {64, 34, 25, 12, 22, 11, 90};

    // Displaying original array
    cout << "Original array: ";
    for (int num : arr) {
        cout << num << " ";
    }
    cout << endl;

    // Calling bubbleSort function
    bubbleSort(arr);

    // Displaying sorted array
    cout << "Sorted array: ";
    for (int num : arr) {
        cout << num << " ";
    }
    cout << endl;

    return 0;
}

/*
Time Complexity Analysis:
1. Worst-case: O(n^2) - When the array is sorted in reverse order.
2. Best-case: O(n) - When the array is already sorted (Just one pass is needed).
3. Average-case: O(n^2) - When the array is randomly sorted.

Space Complexity Analysis:
1. Space Complexity: O(1) - Sorting is done in-place with constant extra memory.
*/
