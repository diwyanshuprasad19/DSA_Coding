#include <iostream>
#include <vector>
using namespace std;

// Function to perform Selection Sort in descending order
void selectionSort(vector<int>& arr) {
    int n = arr.size();
    
    for (int i = 0; i < n - 1; i++) {
        int maxIndex = i;

        // Find the index of the maximum element in the unsorted part
        for (int j = i + 1; j < n; j++) {
            if (arr[j] > arr[maxIndex]) {
                maxIndex = j;
            }
        }

        // Swap the found maximum element with the first element of the unsorted part
        if (maxIndex != i) {
            swap(arr[i], arr[maxIndex]);
        }
    }
}

int main() {
    vector<int> arr = {64, 34, 25, 12, 22, 11, 90};

    // Displaying original array
    cout << "Original array: ";
    for (int num : arr) {
        cout << num << " ";
    }
    cout << endl;

    // Sorting the array in descending order using Selection Sort
    selectionSort(arr);

    // Displaying sorted array
    cout << "Sorted array in descending order: ";
    for (int num : arr) {
        cout << num << " ";
    }
    cout << endl;

    return 0;
}
