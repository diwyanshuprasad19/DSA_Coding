Sol:
import sys
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int) -> int:
        n = len(nums)
        ans = sys.maxsize

        # prefix sum map
        # key = sum from left
        # value = number of operations from left
        prefix = {0: 0}

        curr = 0
        for i in range(n):
            curr += nums[i]

            # if left side alone becomes x
            if curr == x:
                ans = min(ans, i + 1)

            # store prefix sum with operations count
            prefix[curr] = i + 1

        # now check right side
        curr = 0
        for j in range(n - 1, -1, -1):
            curr += nums[j]
            right_ops = n - j

            # if right side alone becomes x
            if curr == x:
                ans = min(ans, right_ops)

            remaining = x - curr

            # need this much from left side
            if remaining in prefix:
                left_ops = prefix[remaining]

                # important: left and right should not overlap
                if left_ops + right_ops <= n:
                    ans = min(ans, left_ops + right_ops)

        if ans == sys.maxsize:
            return -1

        return ans
