Code:

#include <bits/stdc++.h>
using namespace std;

int solve(int i, int j, vector<vector<int>> &matrix, int n, int m) {
    if (j < 0 || j >= m) return -1e9;
    if (i == 0) return matrix[i][j];

    int down = matrix[i][j] + solve(i - 1, j, matrix, n, m);
    int ld = matrix[i][j] + solve(i - 1, j - 1, matrix, n, m);
    int rd = matrix[i][j] + solve(i - 1, j + 1, matrix, n, m);

    return max({down, ld, rd});
}

int getMaxPathSum(vector<vector<int>> &matrix) {
    int n = matrix.size();
    int m = matrix[0].size();
    int result = INT_MIN;

    for (int j = 0; j < m; j++) {
        result = max(result, solve(n - 1, j, matrix, n, m));
    }

    return result;
}

int main() {
    vector<vector<int>> matrix = {
        {10, 10, 2, 0, 20, 4},
        {1, 0, 0, 30, 2, 5},
        {0, 10, 4, 0, 2, 0},
        {1, 0, 2, 20, 0, 4}
    };
    cout << getMaxPathSum(matrix) << endl;  // Output: 74
    return 0;
}

🔸 Explanation:
One possible path: matrix[0][4] → matrix[1][3] → matrix[2][3] → matrix[3][3]
→ 20 + 30 + 0 + 20 = 70, or a better one: 20 + 30 + 0 + 24 = 74

Time and Space Complexity
⏱ Time Complexity: O(3^n)
Each recursive call branches into 3 subproblems: down, left-diagonal, right-diagonal

For a matrix of size n x m, time is exponential → O(m × 3^n)
(as we try all paths starting from m columns)

💾 Space Complexity: O(n) (recursion stack)
