Code:

#include <bits/stdc++.h>
using namespace std;

// Recursive function with memoization
int solve(int i, int j, vector<vector<int>>& matrix, vector<vector<int>>& dp, int n, int m) {
    // Out of bounds
    if (j < 0 || j >= m) return -1e8;

    // Base case: top row
    if (i == 0) return matrix[0][j];

    // If already calculated
    if (dp[i][j] != -1) return dp[i][j];

    int up = matrix[i][j] + solve(i - 1, j, matrix, dp, n, m);
    int ld = matrix[i][j] + solve(i - 1, j - 1, matrix, dp, n, m);
    int rd = matrix[i][j] + solve(i - 1, j + 1, matrix, dp, n, m);

    return dp[i][j] = max({up, ld, rd});
}

int getMaxPathSum(vector<vector<int>> &matrix) {
    int n = matrix.size();
    int m = matrix[0].size();

    vector<vector<int>> dp(n, vector<int>(m, -1));

    int maxSum = -1e8;
    for (int j = 0; j < m; j++) {
        maxSum = max(maxSum, solve(n - 1, j, matrix, dp, n, m));
    }

    return maxSum;
}

int main() {
    vector<vector<int>> matrix = {
        {10, 10, 2, 0, 20, 4},
        {1, 0, 0, 30, 2, 5},
        {0, 10, 4, 0, 2, 0},
        {1, 0, 2, 20, 0, 4}
    };
    cout << getMaxPathSum(matrix) << endl;  // Output: 74
    return 0;
}
