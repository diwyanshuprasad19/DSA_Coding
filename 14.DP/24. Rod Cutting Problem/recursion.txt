Code:

#include <bits/stdc++.h>
using namespace std;

int solve(int index, int length, vector<int> &price) {
    if (index == 0) {
        return length * price[0];  // take full length with unit price
    }

    int notTake = solve(index - 1, length, price);
    int take = INT_MIN;
    int rodLength = index + 1;

    if (rodLength <= length)
        take = price[index] + solve(index, length - rodLength, price);

    return max(take, notTake);
}

int cutRod(vector<int> &price, int n) {
    return solve(n - 1, n, price);
}

int main() {
    vector<int> price = {2, 5, 7, 8};
    int n = 4;
    cout << cutRod(price, n) << endl;  // Output: 10
    return 0;
}
