int longestBitonicSequence(vector<int>& arr, int n) {
    vector<int> dp1(n,1), dp2(n,1);

    // Step 1: Compute LIS from left to right
    for(int i=0;i<n;i++) {
        for(int j=0;j<i;j++) {
            if(arr[j]<arr[i] && 1+dp1[j]>dp1[i])
                dp1[i]=1+dp1[j];
        }
    }

    // Step 2: Compute LIS from right to left (equivalent to LDS)
    reverse(arr.begin(),arr.end());
    for(int i=0;i<n;i++) {
        for(int j=0;j<i;j++) {
            if(arr[j]<arr[i] && 1+dp2[j]>dp2[i])
                dp2[i]=1+dp2[j];
        }
    }
    reverse(dp2.begin(),dp2.end());

    // Step 3: Combine LIS and LDS for bitonic length
    int maxi=0;
    for(int i=0;i<n;i++) {
        maxi=max(maxi,dp1[i]+dp2[i]-1);
    }

    return maxi;
}


Time & Space Complexity
Time Complexity: O(n^2)
(Two passes for LIS and one for LDS using nested loops)

Space Complexity: O(n)
(Two DP arrays dp1 and dp2)
