#include <bits/stdc++.h>
using namespace std;

bool solve(int index, int target, vector<int> &arr, vector<vector<int>> &dp) {
    if (target == 0) return true;
    if (index == 0) return arr[0] == target;

    if (dp[index][target] != -1) return dp[index][target];

    bool notTake = solve(index - 1, target, arr, dp);
    bool take = false;
    if (target >= arr[index])
        take = solve(index - 1, target - arr[index], arr, dp);

    return dp[index][target] = take || notTake;
}

bool subsetSumToK(int n, int k, vector<int> &arr) {
    vector<vector<int>> dp(n, vector<int>(k + 1, -1));
    return solve(n - 1, k, arr, dp);
}

int main() {
    vector<int> arr = {2, 3, 1, 5};
    int k = 4;
    cout << (subsetSumToK(arr.size(), k, arr) ? "true" : "false") << endl;
    return 0;
}
