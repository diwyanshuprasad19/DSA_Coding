#include <bits/stdc++.h>
using namespace std;

int solve(int index, int prevIndex, int arr[], int n) {
    if (index == n) return 0;

    int notTake = solve(index + 1, prevIndex, arr, n);

    int take = 0;
    if (prevIndex == -1 || arr[index] > arr[prevIndex]) {
        take = 1 + solve(index + 1, index, arr, n);
    }

    return max(take, notTake);
}

int longestIncreasingSubsequence(int arr[], int n) {
    return solve(0, -1, arr, n);
}

int main() {
    int arr[] = {10, 9, 2, 5, 3, 7, 101, 18};
    int n = sizeof(arr) / sizeof(arr[0]);

    cout << "Length of LIS: " << longestIncreasingSubsequence(arr, n) << endl;
    return 0;
}
