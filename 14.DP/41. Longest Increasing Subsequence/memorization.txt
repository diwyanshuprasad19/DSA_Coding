#include <bits/stdc++.h>
using namespace std;

int solve(int index, int prevIndex, int arr[], int n, vector<vector<int>> &dp) {
    if (index == n) return 0;

    if (dp[index][prevIndex + 1] != -1)
        return dp[index][prevIndex + 1]; // ✅ Fix: access using prevIndex + 1

    int notTake = solve(index + 1, prevIndex, arr, n, dp);

    int take = 0;
    if (prevIndex == -1 || arr[index] > arr[prevIndex])
        take = 1 + solve(index + 1, index, arr, n, dp);

    return dp[index][prevIndex + 1] = max(take, notTake);
}

int longestIncreasingSubsequence(int arr[], int n) {
    vector<vector<int>> dp(n, vector<int>(n + 1, -1)); // prevIndex ranges from -1 to n-1
    return solve(0, -1, arr, n, dp);
}

int main() {
    int arr[] = {10, 9, 2, 5, 3, 7, 101, 18};
    int n = sizeof(arr) / sizeof(arr[0]);

    cout << "Length of LIS: " << longestIncreasingSubsequence(arr, n) << endl;
    return 0;
}
