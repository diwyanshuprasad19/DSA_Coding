Code:

#include <bits/stdc++.h>
using namespace std;

int solve(int index, int target, vector<int> &num) {
    if (target == 0) return 1;
    if (index == 0) return num[0] == target;

    int notPick = solve(index - 1, target, num);
    int pick = 0;
    if (num[index] <= target)
        pick = solve(index - 1, target - num[index], num);

    return pick + notPick;
}

int findWays(vector<int> &num, int tar) {
    return solve(num.size() - 1, tar, num);
}

int main() {
    vector<int> num = {1, 2, 2, 3};
    int tar = 4;
    cout << findWays(num, tar) << endl;  // Output: 3
    return 0;
}
