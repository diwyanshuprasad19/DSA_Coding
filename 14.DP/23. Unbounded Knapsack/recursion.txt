Code:

#include <bits/stdc++.h>
using namespace std;

int solve(int index, int maxweight, vector<int> &profit, vector<int> &weight) {
    if (index == 0) {
        // You can take as many as fit of item 0
        return (maxweight / weight[0]) * profit[0];
    }

    int notTake = solve(index - 1, maxweight, profit, weight);
    int take = INT_MIN;
    if (weight[index] <= maxweight)
        take = profit[index] + solve(index, maxweight - weight[index], profit, weight);

    return max(take, notTake);
}

int unboundedKnapsack(int n, int w, vector<int> &profit, vector<int> &weight) {
    return solve(n - 1, w, profit, weight);
}

int main() {
    vector<int> profit = {10, 30, 20};
    vector<int> weight = {5, 10, 15};
    int w = 100;
    cout << unboundedKnapsack(profit.size(), w, profit, weight) << endl;  // Output: 300
    return 0;
}
