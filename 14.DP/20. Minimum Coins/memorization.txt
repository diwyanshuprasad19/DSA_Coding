#include <bits/stdc++.h>
using namespace std;

int solve(int index, vector<int>& num, int x, vector<vector<int>>& dp) {
    if (index == 0) {
        if (x % num[0] == 0)
            return x / num[0];
        else
            return 1e9;  // very large, means invalid
    }

    if (dp[index][x] != -1)
        return dp[index][x];

    int notTake = solve(index - 1, num, x, dp);
    int take = 1e9;
    if (num[index] <= x)
        take = 1 + solve(index, num, x - num[index], dp);

    return dp[index][x] = min(take, notTake);
}

int minimumElements(vector<int>& num, int x) {
    int n = num.size();
    vector<vector<int>> dp(n, vector<int>(x + 1, -1));

    int result = solve(n - 1, num, x, dp);
    return (result >= 1e9) ? -1 : result;
}

int main() {
    vector<int> num = {1, 2, 3};
    int x = 7;
    cout << minimumElements(num, x) << endl;  // Output: 3
    return 0;
}
