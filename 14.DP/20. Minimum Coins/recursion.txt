#include <bits/stdc++.h>
using namespace std;

int solve(int index, vector<int>& num, int x) {
    if (index == 0) {
        if (x % num[0] == 0) return x / num[0];
        return 1e9;  // can't make amount with this coin
    }

    int notTake = solve(index - 1, num, x);  // don't use this coin
    int take = 1e9;
    if (num[index] <= x)
        take = 1 + solve(index, num, x - num[index]);  // use this coin

    return min(take, notTake);
}

int minimumElements(vector<int>& num, int x) {
    int count = solve(num.size() - 1, num, x);
    return (count >= 1e9) ? -1 : count;
}

int main() {
    vector<int> num = {1, 2, 3};
    int x = 7;
    cout << minimumElements(num, x) << endl;  // Output: 3 (2+2+3 or 1+1+1+1+1+1+1)
    return 0;
}
