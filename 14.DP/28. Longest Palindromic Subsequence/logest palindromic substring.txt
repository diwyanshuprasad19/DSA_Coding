#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    string longestPalindrome(string s) {
        int start = 0, end = 0;

        for(int i = 0; i < s.length(); i++) {
            int len1 = expandCenter(s, i, i);       // Odd length palindrome
            int len2 = expandCenter(s, i, i + 1);   // Even length palindrome
            int len = max(len1, len2);

            if(len > end - start) {
                start = i - (len - 1) / 2;
                end = i + len / 2;
            }
        }

        return s.substr(start, end - start + 1);
    }

    int expandCenter(string s, int left, int right) {
        while(left >= 0 && right < s.length() && s[left] == s[right]) {
            left--;
            right++;
        }
        return right - left - 1;
    }
};

int main() {
    Solution sol;
    
    string input = "babad";
    string result = sol.longestPalindrome(input);

    cout << "Longest Palindromic Substring of \"" << input << "\" is: " << result << endl;

    return 0;
}
