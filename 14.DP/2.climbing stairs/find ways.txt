def climb_stairs_helper(n, dp):
    if n <= 2:
        return n
    if dp[n] != -1:
        return dp[n]
    dp[n] = climb_stairs_helper(n - 1, dp) + climb_stairs_helper(n - 2, dp)
    return dp[n]


def climb_stairs(n):
    dp = [-1] * (n + 1)
    return climb_stairs_helper(n, dp)


# Direct call (like C++ main logic)
n = 5
print("Ways to climb", n, "stairs:", climb_stairs(n))
