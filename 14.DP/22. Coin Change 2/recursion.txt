Code:

#include <bits/stdc++.h>
using namespace std;

int solve(int index, int target, int *deno) {
    if (index == 0) {
        return (target % deno[0] == 0) ? 1 : 0;
    }

    int notTake = solve(index - 1, target, deno);
    int take = 0;
    if (deno[index] <= target)
        take = solve(index, target - deno[index], deno);

    return notTake + take;
}

long countWaysToMakeChange(int *denominations, int n, int value) {
    return solve(n - 1, value, denominations);
}

int main() {
    int denominations[] = {1, 2, 3};
    int n = 3, value = 4;
    cout << countWaysToMakeChange(denominations, n, value) << endl;  // Output: 4
    return 0;
}
