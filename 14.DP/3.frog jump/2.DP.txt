#include <bits/stdc++.h>
using namespace std;

// Top-down recursive function with memoization
int frogJumpHelper(int n, vector<int> &heights, vector<int> &dp) {
    if (n == 0) return 0;  // base case

    if (dp[n] != -1) return dp[n];  // return cached result

    int left = frogJumpHelper(n - 1, heights, dp) + abs(heights[n] - heights[n - 1]);
    int right = INT_MAX;
    if (n > 1)
        right = frogJumpHelper(n - 2, heights, dp) + abs(heights[n] - heights[n - 2]);

    return dp[n] = min(left, right);  // store and return result
}

int frogJump(int n, vector<int> &heights) {
    vector<int> dp(n, -1);
    return frogJumpHelper(n - 1, heights, dp);
}

int main() {
    vector<int> heights = {10, 20, 30, 10};
    int n = heights.size();

    int result = frogJump(n, heights);
    cout << "Minimum energy required: " << result << endl;

    return 0;
}
