def solve(index, nums):
    if index == 0:
        return nums[index]
    elif index < 0:
        return 0

    pick = nums[index] + solve(index - 2, nums)
    not_pick = solve(index - 1, nums)

    return max(pick, not_pick)


def maximum_non_adjacent_sum(nums):
    k = len(nums)
    return solve(k - 1, nums)


# -------------------------------
# Driver code (same as C++)
# -------------------------------
v = [1, 2, 3, 4]
result = maximum_non_adjacent_sum(v)
print(result)

🧪 Sample Input
v = [1, 2, 3, 4]

✅ Sample Output
6

🧠 Explanation

Possible non-adjacent selections:

1 + 3 = 4

2 + 4 = 6 ✅ (maximum)

So the result is 6.
