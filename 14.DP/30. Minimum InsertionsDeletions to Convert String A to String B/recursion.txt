Code:

#include <iostream>
using namespace std;

int lcs(string &s1, string &s2, int m, int n) {
    if (m == 0 || n == 0)
        return 0;

    if (s1[m - 1] == s2[n - 1])
        return 1 + lcs(s1, s2, m - 1, n - 1);
    else
        return max(lcs(s1, s2, m, n - 1), lcs(s1, s2, m - 1, n));
}

int minOperations(string s1, string s2) {
    int m = s1.size();
    int n = s2.size();

    int len = lcs(s1, s2, m, n);
    int minDeletions = m - len;
    int minInsertions = n - len;

    return minDeletions + minInsertions;
}

int main() {
    string s1 = "AGGTAB";
    string s2 = "GXTXAYB";
    int res = minOperations(s1, s2);
    cout << "Minimum Insertions and Deletions: " << res << endl;
    return 0;
}

Explanation:
LCS = "GTAB" of length 4

Deletions = 6 - 4 = 2

Insertions = 7 - 4 = 3

Total = 2 + 3 = 5
