#include <bits/stdc++.h>
using namespace std;

int solve(int i, int j, string &s1, string &s2, vector<vector<int>> &dp) {
    if (i < 0 || j < 0)
        return 0;

    if (dp[i][j] != -1)
        return dp[i][j];

    if (s1[i] == s2[j])
        return dp[i][j] = 1 + solve(i - 1, j - 1, s1, s2, dp);

    return dp[i][j] = max(solve(i - 1, j, s1, s2, dp), solve(i, j - 1, s1, s2, dp));
}

void lcs(string s1, string s2) {
    int n = s1.length(), m = s2.length();
    vector<vector<int>> dp(n, vector<int>(m, -1));

    // Fill dp using memoization
    solve(n - 1, m - 1, s1, s2, dp);

    // Step 2: Reconstruct LCS using filled dp table
    int len = dp[n - 1][m - 1];
    string str(len, '$');
    int index = len - 1;
    int i = n - 1, j = m - 1;

    while (i >= 0 && j >= 0) {
        if (s1[i] == s2[j]) {
            str[index] = s1[i];
            index--;
            i--;
            j--;
        } else if (i > 0 && (j == 0 || dp[i - 1][j] > dp[i][j - 1])) {
            i--;
        } else {
            j--;
        }
    }

    cout << str << endl;
}

int main() {
    string s1 = "abcde";
    string s2 = "bdgek";

    cout << "The Longest Common Subsequence is: ";
    lcs(s1, s2);  // Output: "bde"
    return 0;
}
