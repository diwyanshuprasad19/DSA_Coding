Code:

#include <bits/stdc++.h>
using namespace std;

bool solve(int i, int j, string &s1, string &s2) {
    if (i < 0 && j < 0) return true;
    if (i < 0 && j >= 0) return false;
    if (j < 0 && i >= 0) {
        for (int k = 0; k <= i; k++) {
            if (s1[k] != '*') return false;
        }
        return true;
    }

    if (s1[i] == s2[j] || s1[i] == '?') {
        return solve(i - 1, j - 1, s1, s2);
    } else if (s1[i] == '*') {
        // '*' can act as empty (i-1, j) or match one (i, j-1)
        return solve(i - 1, j, s1, s2) || solve(i, j - 1, s1, s2);
    } else {
        return false;
    }
}

bool wildcardMatching(string pattern, string text) {
    int n = pattern.length();
    int m = text.length();
    return solve(n - 1, m - 1, pattern, text);
}

int main() {
    string pattern = "a*b?d";
    string text = "aaabxd";
    cout << (wildcardMatching(pattern, text) ? "Match" : "No Match") << endl;
    return 0;
}


Time and Space Complexity
Type	Complexity
Time (worst)	O(2^(n+m)) (due to branching recursion)
Space (stack)	O(n + m) (recursion depth)
