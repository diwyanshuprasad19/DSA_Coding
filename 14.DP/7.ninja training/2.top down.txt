#include <bits/stdc++.h>
using namespace std;

int solve(int index, int last, vector<vector<int>> &points, vector<vector<int>> &dp)
{
    if (index == 0)
    {
        int maxi = INT_MIN;
        for (int i = 0; i < 3; i++)
        {
            if (i != last)
                maxi = max(maxi, points[0][i]);
        }
        return dp[0][last] = maxi;
    }

    if (dp[index][last] != -1) return dp[index][last];

    int maxi = INT_MIN;
    for (int i = 0; i < 3; i++)
    {
        if (i != last)
        {
            int p = points[index][i] + solve(index - 1, i, points, dp);
            maxi = max(maxi, p);
        }
    }

    return dp[index][last] = maxi;
}

int ninjaTraining(int n, vector<vector<int>> &points)
{
    vector<vector<int>> dp(n, vector<int>(4, -1));
    return solve(n - 1, 3, points, dp);
}

int main()
{
    vector<vector<int>> points = {
        {10, 50, 1},
        {5, 100, 11},
        {20, 5, 90}
    };

    int n = points.size();
    int result = ninjaTraining(n, points);
    cout << "Maximum Points: " << result << endl;

    return 0;
}
