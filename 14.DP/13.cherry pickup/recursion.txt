Code:

#include <bits/stdc++.h>
using namespace std;

int solve(int i, int j1, int j2, int row, int col, vector<vector<int>> &grid) {
    if (j1 < 0 || j1 >= col || j2 < 0 || j2 >= col)
        return -1e9;

    if (i == row - 1) {
        if (j1 == j2) return grid[i][j1];
        else return grid[i][j1] + grid[i][j2];
    }

    int maxi = INT_MIN;
    for (int d1 = -1; d1 <= 1; d1++) {
        for (int d2 = -1; d2 <= 1; d2++) {
            int val;
            if (j1 == j2)
                val = grid[i][j1] + solve(i + 1, j1 + d1, j2 + d2, row, col, grid);
            else
                val = grid[i][j1] + grid[i][j2] + solve(i + 1, j1 + d1, j2 + d2, row, col, grid);

            maxi = max(maxi, val);
        }
    }

    return maxi;
}

int maximumChocolates(int r, int c, vector<vector<int>> &grid) {
    return solve(0, 0, c - 1, r, c, grid);
}

int main() {
    vector<vector<int>> grid = {
        {3, 1, 1},
        {2, 5, 1},
        {1, 5, 5},
        {2, 1, 1}
    };
    cout << maximumChocolates(4, 3, grid) << endl;  // Output: 24
    return 0;
}

🔸 Explanation:
Path 1: (0,0) → (1,0) → (2,1) → (3,1)
Path 2: (0,2) → (1,2) → (2,2) → (3,2)

Total = 3 + 1 + 2 + 5 + 5 + 5 + 1 + 2 = 24

✅ Time and Space Complexity (Pure Recursion)
Metric	Value
⏱ Time Complexity	O(3^r * 3^r) = O(9^r) exponential
💾 Space Complexity	O(r) (stack depth)

Why? Each friend has 3 choices → 3×3 = 9 states per row.
