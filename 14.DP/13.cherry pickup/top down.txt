Code:

#include <bits/stdc++.h>
using namespace std;

int solve(int i, int j1, int j2, int row, int col, vector<vector<int>> &grid, vector<vector<vector<int>>> &dp) {
    // Invalid position
    if (j1 < 0 || j1 >= col || j2 < 0 || j2 >= col)
        return -1e9;

    // Base case: last row
    if (i == row - 1) {
        if (j1 == j2) return grid[i][j1];
        return grid[i][j1] + grid[i][j2];
    }

    // Already computed
    if (dp[i][j1][j2] != -1) return dp[i][j1][j2];

    int maxi = INT_MIN;
    for (int d1 = -1; d1 <= 1; d1++) {
        for (int d2 = -1; d2 <= 1; d2++) {
            int val;
            if (j1 == j2)
                val = grid[i][j1] + solve(i + 1, j1 + d1, j2 + d2, row, col, grid, dp);
            else
                val = grid[i][j1] + grid[i][j2] + solve(i + 1, j1 + d1, j2 + d2, row, col, grid, dp);

            maxi = max(maxi, val);
        }
    }

    return dp[i][j1][j2] = maxi;
}

int maximumChocolates(int r, int c, vector<vector<int>> &grid) {
    vector<vector<vector<int>>> dp(r, vector<vector<int>>(c, vector<int>(c, -1)));
    return solve(0, 0, c - 1, r, c, grid, dp);
}

int main() {
    vector<vector<int>> grid = {
        {3, 1, 1},
        {2, 5, 1},
        {1, 5, 5},
        {2, 1, 1}
    };
    int r = 4, c = 3;
    cout << maximumChocolates(r, c, grid) << endl;  // Output: 24
    return 0;
}

🔸 Explanation:
Friend 1: (0, 0) → (1, 0) → (2, 1) → (3, 1)

Friend 2: (0, 2) → (1, 2) → (2, 2) → (3, 2)

Collected = 3 + 1 + 2 + 5 + 5 + 5 + 1 + 2 = 24

✅ Time and Space Complexity
Metric	Value
⏱ Time Complexity	O(r * c * c * 9) = O(r * c²)
💾 Space Complexity	O(r * c * c) (3D DP table) + O(r) (call stack)

Why O(r * c * c)?
We use a 3D DP:

dp[i][j1][j2] stores the max chocolates collected starting from row i with friend1 at column j1 and friend2 at column j2.
