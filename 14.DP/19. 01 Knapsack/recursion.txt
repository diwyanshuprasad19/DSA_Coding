#include <bits/stdc++.h>
using namespace std;

int solve(int index, vector<int>& weight, vector<int>& value, int maxWeight) {
    // Base case: first item
    if (index == 0) {
        if (weight[0] <= maxWeight)
            return value[0];
        else
            return 0;
    }

    // Not taking the item
    int notTake = solve(index - 1, weight, value, maxWeight);

    // Taking the item (if it fits)
    int take = INT_MIN;
    if (weight[index] <= maxWeight)
        take = value[index] + solve(index - 1, weight, value, maxWeight - weight[index]);

    return max(take, notTake);
}

int knapsack(vector<int> weight, vector<int> value, int n, int maxWeight) {
    return solve(n - 1, weight, value, maxWeight);
}

int main() {
    vector<int> weight = {1, 2, 4, 5};
    vector<int> value = {5, 4, 8, 6};
    int maxWeight = 5;
    cout << knapsack(weight, value, weight.size(), maxWeight) << endl;  // Output: 13
    return 0;
}
